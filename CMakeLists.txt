cmake_minimum_required(VERSION 3.16)
project(journaled_flash_store C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(store src/store.c)
target_include_directories(store PUBLIC include)
target_compile_options(store PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(flash_model sim/flash_model.c)
target_include_directories(flash_model PUBLIC sim)
target_link_libraries(flash_model PUBLIC store)
target_compile_options(flash_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(flash_store_sim sim/main.c)
target_link_libraries(flash_store_sim PRIVATE flash_model)
add_executable(flash_store_tests tests/test_main.c)
target_link_libraries(flash_store_tests PRIVATE flash_model)
enable_testing()
add_test(NAME flash_store_tests COMMAND flash_store_tests)
